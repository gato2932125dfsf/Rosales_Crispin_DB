-- 1. CREACIÓN DE LA BASE DE DATOS
CREATE DATABASE GestionInventarioTecnologico;
GO

USE GestionInventarioTecnologico;
GO

-- ========================================================
-- 2. CREACIÓN DE TABLAS INDEPENDIENTES (FUERTES / CATÁLOGOS)
-- ========================================================

CREATE TABLE Procesador (
    Codigo_Procesador VARCHAR(50) NOT NULL,
    Procesador_Velocidad VARCHAR(50),
    CONSTRAINT PK_Procesador PRIMARY KEY (Codigo_Procesador)
);

CREATE TABLE Disco_Duro (
    ID_DiscoDuro VARCHAR(50) NOT NULL,
    DiscoDuro_Capacidad VARCHAR(50),
    DiscoDuro_Marca VARCHAR(50),
    CONSTRAINT PK_Disco_Duro PRIMARY KEY (ID_DiscoDuro)
);

CREATE TABLE GPU (
    Codigo_GPU VARCHAR(50) NOT NULL,
    GPU_Memoria VARCHAR(50),
    CONSTRAINT PK_GPU PRIMARY KEY (Codigo_GPU)
);

CREATE TABLE Ficha_Inventario_Tecnologico (
    Id_CodigoAlmacen VARCHAR(50) NOT NULL,
    Dependencia VARCHAR(100),
    Fecha DATE,
    Cod_Tecnologia VARCHAR(50),
    Usuario VARCHAR(100), -- Nota: Aparece duplicado en el diseño, se coloca una sola vez
    CONSTRAINT PK_Ficha_Inventario PRIMARY KEY (Id_CodigoAlmacen)
);

CREATE TABLE Acceso_Remoto (
    Codigo_Anidesk VARCHAR(50) NOT NULL,
    contraseña VARCHAR(50),
    CONSTRAINT PK_Acceso_Remoto PRIMARY KEY (Codigo_Anidesk)
);

CREATE TABLE General (
    ID_General VARCHAR(50) NOT NULL,
    Estado VARCHAR(50),
    Observacion VARCHAR(MAX),
    Firma VARCHAR(100),
    CONSTRAINT PK_General PRIMARY KEY (ID_General)
);

CREATE TABLE Sistema_Operativo (
    ID_Sistema_Operativo VARCHAR(50) NOT NULL,
    Sistema_Operativo VARCHAR(100),
    Sistema_licencia VARCHAR(50), -- Aquí puedes manejar si es Licenciado o Sin Licencia
    CONSTRAINT PK_Sistema_Operativo PRIMARY KEY (ID_Sistema_Operativo)
);

CREATE TABLE Monitor (
    N_Serie_Monitor VARCHAR(50) NOT NULL,
    modelo VARCHAR(50),
    marca VARCHAR(50),
    CONSTRAINT PK_Monitor PRIMARY KEY (N_Serie_Monitor)
);

CREATE TABLE Teclado (
    N_Serie_Teclado VARCHAR(50) NOT NULL,
    modelo VARCHAR(50),
    CONSTRAINT PK_Teclado PRIMARY KEY (N_Serie_Teclado)
);

CREATE TABLE Mouse (
    N_Serie_Mouse VARCHAR(50) NOT NULL,
    modelo VARCHAR(50),
    CONSTRAINT PK_Mouse PRIMARY KEY (N_Serie_Mouse)
);

CREATE TABLE Parlante (
    N_Serie_Parlante VARCHAR(50) NOT NULL,
    modelo VARCHAR(50),
    CONSTRAINT PK_Parlante PRIMARY KEY (N_Serie_Parlante)
);
go

-- ========================================================
-- 3. CREACIÓN DE TABLAS CON DEPENDENCIAS (LLAVES FORÁNEAS)
-- ========================================================

-- Tabla Harware (Depende de Procesador, Disco_Duro y GPU)
CREATE TABLE Harware (
    ID_Harware VARCHAR(50) NOT NULL,
    Codigo_Procesador VARCHAR(50),
    ID_DiscoDuro VARCHAR(50),
    Codigo_GPU VARCHAR(50),
    tipo VARCHAR(50),
    Marca_y_Modelo VARCHAR(100),
    N_Serie VARCHAR(50),
    Ram VARCHAR(50),
    Unidad_CD_DVDRW VARCHAR(50),
    Regulador VARCHAR(50),
    CONSTRAINT PK_Harware PRIMARY KEY (ID_Harware),
    CONSTRAINT FK_Harware_Procesador FOREIGN KEY (Codigo_Procesador) 
        REFERENCES Procesador(Codigo_Procesador),
    CONSTRAINT FK_Harware_Disco FOREIGN KEY (ID_DiscoDuro) 
        REFERENCES Disco_Duro(ID_DiscoDuro),
    CONSTRAINT FK_Harware_GPU FOREIGN KEY (Codigo_GPU) 
        REFERENCES GPU(Codigo_GPU)
);

-- Tabla Sofware (Depende de Sistema_Operativo)
-- Nota: Corregido el diseño para que ID_Sofware sea PK única y el SO sea la FK relacional.
CREATE TABLE Sofware (
    ID_Sofware VARCHAR(50) NOT NULL,
    ID_Sistema_Operativo VARCHAR(50), 
    Ofimatica VARCHAR(100),
    Antivirus VARCHAR(100),
    Navegador VARCHAR(100),
    Desarollo VARCHAR(100),
    Diseño VARCHAR(100),
    CONSTRAINT PK_Sofware PRIMARY KEY (ID_Sofware),
    CONSTRAINT FK_Sofware_SO FOREIGN KEY (ID_Sistema_Operativo) 
        REFERENCES Sistema_Operativo(ID_Sistema_Operativo)
);

-- Tabla Accesorio_partes (Depende de Monitor, Teclado, Mouse y Parlante)
CREATE TABLE Accesorio_partes (
    ID_Partes VARCHAR(50) NOT NULL,
    N_Serie_Monitor VARCHAR(50),
    N_Serie_Teclado VARCHAR(50),
    N_Serie_Mouse VARCHAR(50),
    N_Serie_Parlante VARCHAR(50),
    CONSTRAINT PK_Accesorio_partes PRIMARY KEY (ID_Partes),
    CONSTRAINT FK_Partes_Monitor FOREIGN KEY (N_Serie_Monitor) 
        REFERENCES Monitor(N_Serie_Monitor),
    CONSTRAINT FK_Partes_Teclado FOREIGN KEY (N_Serie_Teclado) 
        REFERENCES Teclado(N_Serie_Teclado),
    CONSTRAINT FK_Partes_Mouse FOREIGN KEY (N_Serie_Mouse) 
        REFERENCES Mouse(N_Serie_Mouse),
    CONSTRAINT FK_Partes_Parlante FOREIGN KEY (N_Serie_Parlante) 
        REFERENCES Parlante(N_Serie_Parlante)
);
go

-- ========================================================
-- 4. TABLA CENTRAL: Equipo_Inventario
-- ========================================================
-- Esta tabla une los flujos principales del inventario tecnológico.

CREATE TABLE Equipo_Inventario (
    Id_Inventario VARCHAR(50) NOT NULL,
    Id_CodigoAlmacen VARCHAR(50),
    ID_Harware VARCHAR(50),
    ID_Sofware VARCHAR(50),
    ID_General VARCHAR(50),
    ID_Partes VARCHAR(50),
    Codigo_Anidesk VARCHAR(50),
    CONSTRAINT PK_Equipo_Inventario PRIMARY KEY (Id_Inventario),
    CONSTRAINT FK_Equipo_Ficha FOREIGN KEY (Id_CodigoAlmacen) 
        REFERENCES Ficha_Inventario_Tecnologico(Id_CodigoAlmacen),
    CONSTRAINT FK_Equipo_Harware FOREIGN KEY (ID_Harware) 
        REFERENCES Harware(ID_Harware),
    CONSTRAINT FK_Equipo_Sofware FOREIGN KEY (ID_Sofware) 
        REFERENCES Sofware(ID_Sofware),
    CONSTRAINT FK_Equipo_General FOREIGN KEY (ID_General) 
        REFERENCES General(ID_General),
    CONSTRAINT FK_Equipo_Partes FOREIGN KEY (ID_Partes) 
        REFERENCES Accesorio_partes(ID_Partes),
    CONSTRAINT FK_Equipo_Anidesk FOREIGN KEY (Codigo_Anidesk) 
        REFERENCES Acceso_Remoto(Codigo_Anidesk)
);
GO

USE GestionInventarioTecnologico;
GO

-- ========================================================
-- Proyecto 7 – Multiplicación de capacidad de disco
-- ========================================================
-- Enunciado: Calcular el doble de almacenamiento.
-- Verificación:
SELECT * FROM Disco_Duro;
-- Consulta del ejercicio:
SELECT 
    ID_DiscoDuro,
    DiscoDuro_Capacidad AS Capacidad_Original,
    CASE 
        WHEN DiscoDuro_Capacidad LIKE '%GB%' THEN 
            CONCAT(CAST(LEFT(DiscoDuro_Capacidad, CHARINDEX('GB', DiscoDuro_Capacidad) - 1) AS INT) * 2, 'GB')
        WHEN DiscoDuro_Capacidad LIKE '%TB%' THEN 
            CONCAT(CAST(LEFT(DiscoDuro_Capacidad, CHARINDEX('TB', DiscoDuro_Capacidad) - 1) AS INT) * 2, 'TB')
        ELSE 'No_Calculable'
    END AS Doble_Almacenamiento
FROM Disco_Duro;
GO

-- ========================================================
-- Proyecto 19 – Uso de AND
-- ========================================================
-- Enunciado: Mostrar laptops HP.
-- Verificación:
SELECT * FROM Equipo_Inventario;
SELECT * FROM Harware;
SELECT * FROM Monitor;
-- Consulta del ejercicio:
SELECT E.Id_Inventario, H.tipo, M.marca
FROM Equipo_Inventario AS E
INNER JOIN Harware AS H ON E.ID_Harware = H.ID_Harware
INNER JOIN Accesorio_partes AS AP ON E.ID_Partes = AP.ID_Partes
INNER JOIN Monitor AS M ON AP.N_Serie_Monitor = M.N_Serie_Monitor
WHERE H.tipo = 'laptop' AND M.marca = 'HP';
GO

-- Proyecto 20 – Uso de OR y paréntesis
-- ========================================================

SELECT * FROM Harware;
SELECT * FROM Monitor;
-- Consulta del ejercicio:
SELECT E.Id_Inventario, H.tipo, M.marca
FROM Equipo_Inventario AS E
INNER JOIN Harware AS H ON E.ID_Harware = H.ID_Harware
INNER JOIN Accesorio_partes AS AP ON E.ID_Partes = AP.ID_Partes
INNER JOIN Monitor AS M ON AP.N_Serie_Monitor = M.N_Serie_Monitor
WHERE (M.marca = 'HP' OR M.marca = 'Dell') 
  AND H.tipo = 'laptop';
